#
# 2026-08-24 프론트/백엔드 분리 결정(docs/decisions.md #2) — STT/TTS/LLM 무거운 GPU
# 추론은 이 프로세스에서 직접 안 돈다. HF Spaces 유료 GPU(T4)에 별도로 떠 있는 백엔드
# (hf_backend/, 그 폴더의 requirements.txt가 무거운 torch/transformers/qwen-tts 스택을
# 담당)를 gradio_client로 원격 호출한다(src/orchestration/inference_backend.py 참고).
# 그래서 이 파일엔 torch/transformers/qwen-tts/faster-whisper가 없다 — 여기 있으면 CPU

#
# 로컬에서 원격 백엔드 없이 STT/TTS/LLM까지 한 프로세스로 다 돌리며 개발/검증할 땐
# requirements-main.txt(env-main, 무거운 전체 스택) 사용.

streamlit==1.61.1

# --- 상시 마이크(실시간 스트리밍 인식, WebRTC) — CPU만으로 동작(GPU 불필요) ---
# src/ui/voice_io.py의 _run_mic_loop()/listen() — streamlit-webrtc가 브라우저 마이크를
# 끊김 없이 스트리밍해주고, silero-vad가 발화 구간을 잘라준다(원본 설계는 proj1-a에서
# 2026-08-23 완성·실사용 검증된 것을 2026-08-24 이 저장소로 그대로 옮겨옴). av/pylibsrtp는
# abi3 안정 ABI 휠이라 Python 버전과 무관하게 프리빌트 휠 설치됨(2026-08-24 PyPI 확인) —
# 소스 빌드 없이 설치될 것으로 예상하나, 실제 Streamlit Cloud 배포에서 최종 확인 필요.
#
# ⚠️ av는 18.1.0으로 처음 고정했다가 실제 배포에서 충돌 남(2026-08-24 실측) —
# streamlit-webrtc가 끌어오는 aiortc==1.14.0/1.15.0이 각각 av<17.0.0 / av<18.0.0을
# 요구해서 av==18.1.0과 동시에 못 깔림(ResolutionImpossible). av PyPI 페이지만 보고
# aiortc 쪽 상한을 확인 안 해서 생긴 실수 — aiortc가 실제로 허용하는 범위 안(<18.0.0)의
# 17.1.0으로 낮춰서 고정한다(이것도 cp311-abi3 휠 있음, 3.14에서도 그대로 설치됨).
streamlit-webrtc==0.77.0
av==17.1.0
# silero-vad가 torch/torchaudio를 끌어오는데, CUDA 지정 없이 설치하면 pip가 CPU용 기본
# 빌드를 받는다(사이즈는 sentence-transformers가 이미 받는 torch와 비슷한 수준) — VAD
# 모델 자체는 아주 작아서(~1MB) 추론 자체는 CPU로도 빠르다.
silero-vad==6.2.1
# ui/mic_vad.py의 MicVadSegmenter._resample()이 librosa.resample()을 쓴다 — 실제 배포
# 로그로 ModuleNotFoundError 확인됨(2026-08-24). requirements-main.txt와 같은 버전으로
# 고정(numba/scipy 등도 딸려오지만 전부 cp314 휠 있음, 2026-08-24 PyPI 확인).
librosa==0.11.0
# 2026-08-24 — 배포 로그로 "Queue overflow"(1024까지 늘려도 계속 넘침) 원인 실측 확인:
# MicVadSegmenter.feed()가 VAD 판정용으로 프레임(20ms)마다 librosa.resample()을 부르는데,
# 이 호출 자체가 프레임 하나당 18.5ms 걸린다(실측) — 초당 50프레임이면 리샘플링만으로
# 이미 실시간의 92%를 써버려 실시간 처리가 구조적으로 불가능했다. librosa는 이미
# soxr을 내부 백엔드로 쓰지만(설치돼 있으면 자동 선택) 매 호출마다 감싸는 검증/설정
# 오버헤드가 이 정도 크기 조각엔 안 맞는다 — soxr을 직접 불러서 감싸는 오버헤드를
# 없애면(quality="QQ", VAD는 에너지 패턴만 보면 돼서 저품질로 충분) 같은 호출이
# 0.009ms로 끝난다(실측, 약 2000배). librosa는 발화 전체를 딱 한 번만 리샘플링하는
# 최종 STT 입력 경로(이미 실사용 검증 완료, 안 건드림)엔 그대로 남겨뒀다 — 이 프레임
# 단위 VAD 경로만 soxr 직접 호출로 바꿔서 librosa의 transitive 의존성이 아니라
# 명시적으로 pin.
soxr==1.1.0

# --- HF Spaces 유료 GPU 백엔드 호출 클라이언트 ---
gradio_client==2.6.0
